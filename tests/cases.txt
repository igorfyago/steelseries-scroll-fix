# Filter cases. One case is four lines:
#   case:   name
#   config: click=0|1  bounce=ms (all buttons)  bounceL/R/M/X1/X2=ms  hold=ms (0 is off)
#           wheel=0|1  reverse=ms  confirm=ms
#   in:     what the mouse sent       "<ms> <L|R|M|X1|X2> <down|up>"  or  "<ms> wheel <+N|-N>"
#   out:    what applications get     same notation, with the time each event is delivered
# An event the filter sends itself is listed at the time it is sent. When it is
# sent because of an incoming event, it comes right after that event.

# ---------- C1: re-press guard ----------
case:   plain click passes
config: click=1 bounce=40 hold=0 wheel=0
in:     0 L down, 80 L up
out:    0 L down, 80 L up

case:   the gap is measured from the last physical release, dropped ones included
